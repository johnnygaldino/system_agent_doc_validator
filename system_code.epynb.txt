import os
import json
from fastapi import FastAPI, Depends, HTTPException, Security
from fastapi.security.api_key import APIKeyHeader
from pydantic import BaseModel, Field
from dotenv import load_dotenv
from openai import AzureOpenAI

# Carrega as variáveis do .env
load_dotenv()

app = FastAPI(
    title="Validador e Tradutor de Documentos Técnicos",
    description="API com FastAPI e Azure OpenAI para traduzir e validar documentações técnicas de forma autenticada.",
    version="1.0.0"
)

# ==========================================
# 1. Configuração de Autenticação (API Key)
# ==========================================
API_KEY = os.getenv("API_SECRET_KEY", "chave-padrao-insegura")
api_key_header = APIKeyHeader(name="X-API-Key", auto_error=True)

def get_api_key(api_key_header: str = Security(api_key_header)):
    """Verifica se a chave enviada no header é válida"""
    if api_key_header != API_KEY:
        raise HTTPException(
            status_code=403, 
            detail="Acesso Negado: Chave de API inválida ou ausente."
        )
    return api_key_header

# ==========================================
# 2. Configuração do Cliente Azure OpenAI
# ==========================================
try:
    client = AzureOpenAI(
        api_key=os.getenv("AZURE_OPENAI_API_KEY"),
        api_version=os.getenv("AZURE_OPENAI_API_VERSION", "2024-02-15-preview"),
        azure_endpoint=os.getenv("AZURE_OPENAI_ENDPOINT")
    )
    DEPLOYMENT_NAME = os.getenv("AZURE_OPENAI_DEPLOYMENT_NAME")
except Exception as e:
    print(f"Aviso: Erro ao configurar o cliente Azure OpenAI. Verifique suas variáveis de ambiente. Detalhes: {e}")
    client = None
    DEPLOYMENT_NAME = None


# ==========================================
# 3. Modelos de Dados (Pydantic)
# ==========================================
class TranslationRequest(BaseModel):
    text: str = Field(..., description="Texto original do documento")
    target_language: str = Field(default="Português", description="Idioma de destino")

class ValidationRequest(BaseModel):
    text: str = Field(..., description="Conteúdo do documento técnico a ser validado")
    doc_type: str = Field(default="Arquitetura de Software", description="Tipo do documento técnico (ex: Manual, API, Arquitetura)")

# ==========================================
# 4. Rotas / Endpoints (Protegidos)
# ==========================================

@app.post("/translate", summary="Traduz um documento")
async def translate_document(req: TranslationRequest, api_key: str = Depends(get_api_key)):
    """Traduz o texto enviado para o idioma desejado usando Azure OpenAI."""
    if not client:
        raise HTTPException(status_code=500, detail="Cliente Azure OpenAI não configurado.")
    try:
        response = client.chat.completions.create(
            model=DEPLOYMENT_NAME,
            messages=[
                {"role": "system", "content": f"Você é um tradutor técnico especialista. Traduza o texto a seguir para {req.target_language}, mantendo o rigor técnico."},
                {"role": "user", "content": req.text}
            ]
        )
        return {"translated_text": response.choices[0].message.content}
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Erro no Azure OpenAI: {str(e)}")


@app.post("/validate", summary="Valida um documento técnico")
async def validate_document(req: ValidationRequest, api_key: str = Depends(get_api_key)):
    """
    Analisa um documento técnico, validando sua coerência, 
    terminologia e formatação. Retorna o resultado em formato JSON estruturado.
    """
    if not client:
        raise HTTPException(status_code=500, detail="Cliente Azure OpenAI não configurado.")
        
    prompt = f"""
    Atue como um Engenheiro Staff sênior fazendo code/doc review.
    Avalie o seguinte documento do tipo '{req.doc_type}'.
    
    Verifique:
    1. Clareza e coesão técnica.
    2. Uso correto da terminologia.
    3. Pontos de melhoria ou vulnerabilidades descritas.
    
    Responda EXCLUSIVAMENTE em formato JSON com as seguintes chaves:
    - "is_valid": booleano indicando se o documento tem qualidade aceitável
    - "score": número de 0 a 100
    - "feedback_geral": string com a sua avaliação
    - "sugestoes_melhoria": lista de strings com dicas práticas
    
    Texto do documento:
    {req.text}
    """
    
    try:
        response = client.chat.completions.create(
            model=DEPLOYMENT_NAME,
            response_format={ "type": "json_object" }, # Garante saída em JSON
            messages=[
                {"role": "system", "content": "Você é um assistente técnico que sempre responde no formato JSON requerido."},
                {"role": "user", "content": prompt}
            ]
        )
        
        # Converte a string JSON retornada pela LLM em um dicionário Python
        result_json = json.loads(response.choices[0].message.content)
        return result_json
        
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Erro na validação com Azure OpenAI: {str(e)}")

# ==========================================
# 5. Inicialização para desenvolvimento local
# ==========================================
if __name__ == "__main__":
    import uvicorn
    uvicorn.run("main:app", host="0.0.0.0", port=8000, reload=True)
